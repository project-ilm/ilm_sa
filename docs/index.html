/* क्रमबद्धायां सूच्यां द्विमान अन्वेषणम् */
पूर्णाङ्क द्विमान_अन्वेषणम्(पूर्णाङ्क मानानि[], पूर्णाङ्क गणनम्, पूर्णाङ्क कुञ्चिका) {
  पूर्णाङ्क नीचः = 0, उच्चः = गणनम् - 1;
  यावत् (नीचः <= उच्चः) {
    पूर्णाङ्क मध्यः = नीचः + (उच्चः - नीचः) / 2;
    यदि (मानानि[मध्यः] == कुञ्चिका) प्रत्यागम मध्यः;
    यदि (मानानि[मध्यः] < कुञ्चिका) नीचः = मध्यः + 1; अन्यथा उच्चः = मध्यः - 1;
  }
  प्रत्यागम -1;
}
पूर्णाङ्क मुख्यम्(शून्य) {
  पूर्णाङ्क मानानि[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
  प्रारूप_लिख("%d %d\n", द्विमान_अन्वेषणम्(मानानि, 10, 23), द्विमान_अन्वेषणम्(मानानि, 10, 7));
  प्रत्यागम 0;
}

# द्विमान अन्वेषणम्, पाइथन भाषायाम् 
परिभाषा द्विमान_अन्वेषणम्(मानानि, कुञ्चिका):
    नीचः, उच्चः = 0, दीर्घता(मानानि) - 1
    यावत् नीचः <= उच्चः:
        मध्यः = (नीचः + उच्चः) // 2
        यदि मानानि[मध्यः] == कुञ्चिका:
            प्रत्यागम मध्यः
        यदि मानानि[मध्यः] < कुञ्चिका:
            नीचः = मध्यः + 1
        अन्यथा:
            उच्चः = मध्यः - 1
    प्रत्यागम -1

मुद्रय(द्विमान_अन्वेषणम्([2, 5, 8, 12, 16, 23, 38], 23), द्विमान_अन्वेषणम्([2, 5, 8], 7))
